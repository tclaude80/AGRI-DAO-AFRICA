# ============================================
# Commandes SFTP manuelles pour OVH
# ============================================
#
# Connexion:
#   sftp -P 22 kftmgreedt@ssh.cluster127.hosting.ovh.net
#
# Puis exécutez ces commandes une par une:

cd /home/kftmgreedt/www

# Créer les dossiers
mkdir a-propos
mkdir assets
mkdir bourse-plantain
mkdir contact
mkdir durabilite
mkdir export
mkdir filiere
mkdir images
mkdir mentions-legales
mkdir politique-confidentialite
mkdir process-qualite
mkdir produits
mkdir tracabilite

# Uploader les fichiers (depuis le dossier static/dist local)
lcd static/dist

put .htaccess
put .htpasswd
put index.html
put favicon.svg
put robots.txt
put site.webmanifest

put -r assets
put -r a-propos
put -r bourse-plantain
put -r contact
put -r durabilite
put -r export
put -r filiere
put -r images
put -r mentions-legales
put -r politique-confidentialite
put -r process-qualite
put -r produits
put -r tracabilite

bye
